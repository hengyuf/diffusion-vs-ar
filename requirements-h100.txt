# requirements.txt adapted for x86_64 H100 (sm_90, driver 535 / CUDA 12.2).
#

# and sm_100 kernels, so one dependency set covers the H100 and GB200 nodes and
# `uv.lock` resolves unchanged on either architecture.

#   pip install -r requirements-h100.txt --extra-index-url https://download.pytorch.org/whl/cu128

--extra-index-url https://download.pytorch.org/whl/cu128

# cu128 on a 12.2 driver is fine: CUDA minor-version compatibility makes any
# 12.x runtime run on a >= 525 driver. Verified on this node -- see SETUP-h100.md.
